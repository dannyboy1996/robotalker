cmake_minimum_required(VERSION 3.20)
project(RoboTalker CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

if(MSVC)
    add_compile_options(/W4 /permissive-)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Wno-unused-parameter)
endif()

# Static linking on MinGW: bundle libstdc++/libgcc/winpthread into the
# CLI, DLLs, and the GUI so end users don't need MinGW DLLs at runtime.
# The `-static` flag alone forces full static linking of all toolchain
# runtimes (gcc, stdc++, winpthread).
if(MINGW)
    set(STATIC_LINK_FLAGS -static)
endif()

# ---- Embed CMU pronouncing dictionary -------------------------------------
# At configure time, slurp cmudict-0.7b.txt (if present) into a generated
# .cpp file as a raw string literal. The library then ships with the full
# ~134K word dictionary baked in -- no external data file needed at
# runtime. We still fall back to checking the disk paths first so a user
# who wants to override with a custom dict can drop one beside the exe.
set(CMUDICT_SRC "${CMAKE_SOURCE_DIR}/data/cmudict-0.7b.txt")
set(CMUDICT_GEN "${CMAKE_BINARY_DIR}/generated/cmudict_embedded.inc")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
if(EXISTS "${CMUDICT_SRC}")
    file(READ "${CMUDICT_SRC}" CMUDICT_CONTENT)
    # Header-style include containing just the raw data as a static
    # array. text.cpp #includes this directly so there's no linker
    # whole-archive issue when the static library is consumed by the
    # DLL / EXE targets.
    file(WRITE "${CMUDICT_GEN}"
"// AUTO-GENERATED by CMake from data/cmudict-0.7b.txt. Do not edit.\n"
"static const char kEmbeddedCmudict[] = R\"CMUDICT_BAKED(\n${CMUDICT_CONTENT})CMUDICT_BAKED\";\n"
"static const unsigned long kEmbeddedCmudictSize = sizeof(kEmbeddedCmudict) - 1;\n")
    message(STATUS "Embedded CMU dict: ${CMUDICT_SRC}")
else()
    file(WRITE "${CMUDICT_GEN}"
"static const char kEmbeddedCmudict[] = \"\";\n"
"static const unsigned long kEmbeddedCmudictSize = 0;\n")
    message(WARNING "data/cmudict-0.7b.txt not found -- shipping with mini-dict only")
endif()

# ---- Core static library --------------------------------------------------
# Internal: every executable links against this. Kept as a STATIC library so
# the CLI and rvapi DLL can each ship a self-contained build.
add_library(klattalker STATIC
    src/wav.cpp
    src/resonators.cpp
    src/noise.cpp
    src/glottal.cpp
    src/data.cpp
    src/duration.cpp
    src/intensity.cpp
    src/inflection.cpp
    src/sequencer.cpp
    src/engine_klsyn80.cpp
    src/engine_klsyn88.cpp
    src/synth.cpp
    src/text.cpp
    src/espeak_g2p.cpp
    src/japanese_g2p.cpp
    src/japanese_accent_dict.cpp
)
target_include_directories(klattalker PUBLIC include)
target_include_directories(klattalker PRIVATE "${CMAKE_BINARY_DIR}/generated")
set_target_properties(klattalker PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ---- RVAPI: shared library export (the public C SDK) ---------------------
# rvapi = "Robotic Voice API" -- a flat C interface so any language with
# C FFI (Python ctypes, .NET P/Invoke, Rust bindgen, etc.) can drive
# RoboTalker as a library.
add_library(rvapi SHARED src/rvapi.cpp)
target_link_libraries(rvapi PRIVATE klattalker)
target_compile_definitions(rvapi PRIVATE RVAPI_BUILDING)
target_include_directories(rvapi PUBLIC include)
if(MINGW)
    # Static-link MinGW runtime into the DLL so consumers don't need
    # libstdc++-6.dll, libgcc_s_seh-1.dll, libwinpthread-1.dll
    # alongside it. The DLL exports plain C functions.
    target_link_options(rvapi PRIVATE ${STATIC_LINK_FLAGS})
endif()
# Output naming: rvapi.dll on Windows, librvapi.so on Linux
set_target_properties(rvapi PROPERTIES OUTPUT_NAME rvapi)

# ---- CLI: robotalker.exe -------------------------------------------------
add_executable(robotalker apps/cli.cpp)
target_link_libraries(robotalker PRIVATE klattalker)
set_target_properties(robotalker PROPERTIES OUTPUT_NAME robotalker)
if(MINGW)
    target_link_options(robotalker PRIVATE ${STATIC_LINK_FLAGS})
endif()

# ---- GUI: speak_robotalker.exe (Windows-only, accessible Win32) ---------
if(WIN32)
    add_executable(speak_robotalker WIN32 apps/gui.cpp)
    target_link_libraries(speak_robotalker PRIVATE klattalker winmm comctl32)
    if(MINGW)
        # -municode tells MinGW the entry point is wWinMain not WinMain
        target_compile_options(speak_robotalker PRIVATE -municode)
        target_link_options(speak_robotalker PRIVATE -municode ${STATIC_LINK_FLAGS})
    endif()
endif()

# ---- Demo: kept for back-compat ------------------------------------------
add_executable(klattalker_demo examples/demo.cpp)
target_link_libraries(klattalker_demo PRIVATE klattalker)
if(MINGW)
    target_link_options(klattalker_demo PRIVATE ${STATIC_LINK_FLAGS})
endif()
